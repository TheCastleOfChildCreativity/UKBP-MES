  "id" bigint [pk, not null]
  "detail_id" int
  "stage_id" int
}

Table "public"."history_operation" {
  "id" bigint [pk, not null]
  "work_piece" int [not null]
  "stage" int
}

Table "public"."stages" {
  "id" int [pk, not null, increment, ref: - "public"."work_pieces"."current_stage", ref: - "public"."details_stages"."stage_id", ref: - "public"."history_operation"."stage"]
  "name" varchar(255) [not null]
  "created_at" timestamp [not null, default: `CURRENT_TIMESTAMP`]
  "updated_at" timestamp [not null, default: `CURRENT_TIMESTAMP`]
}

Table "public"."details" {
  "id" int [pk, not null, increment, ref: > "public"."details_stages"."detail_id"]
  "name" varchar(255) [not null]
  "type" text
  "created_at" timestamp [not null, default: `CURRENT_TIMESTAMP`]
  "updated_at" timestamp [not null, default: `CURRENT_TIMESTAMP`]
}

Table "public"."work_pieces" {
  "id" bigint [pk, not null]
  "detail" int [not null, ref: > "public"."details"."id", ref: - "public"."history_operation"."work_piece"]
  "current_stage" int [not null]
  "status" varchar(255) [not null]
}
